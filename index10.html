Q4:-Enter a string from user and convert into title case ? 

#include 
#include 

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    gets(str);

    strlwr(str);

    if(str[0] >= 'a' && str[0] <= 'z')
        str[0] = str[0] - 32;

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == ' ' && str[i+1] >= 'a' && str[i+1] <= 'z')
        {
            str[i+1] = str[i+1] - 32;
        }
    }

    printf("Title Case: %s", str);

    return 0;
}
Q5:-Enter a string from user and convert into toggle case ?? 

#include 
#include 

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    scanf("%s", str);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] >= 'a' && str[i] <= 'z')
            str[i] = str[i] - 32;

        else if(str[i] >= 'A' && str[i] <= 'Z')
            str[i] = str[i] + 32;
    }

    printf("Toggle Case: %s", str);

    return 0;
}

Q11:Enter the string from user and count all vowels ?
#include 

int main()
{
    char str[100];
    int i, count = 0;

    printf("Enter a string: ");
    scanf("%s", str);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == 'a' || str[i] == 'e' ||
           str[i] == 'i' || str[i] == 'o' ||
           str[i] == 'u' || str[i] == 'A' ||
           str[i] == 'E' || str[i] == 'I' ||
           str[i] == 'O' || str[i] == 'U')
        {
            count++;
        }
    }

    printf("Total vowels = %d", count);

    return 0;
}
Q12:Enter the string from user and enter a character if it is exist then print the bound number ?

{

    printf("Enter a string: ");
    scanf("%s", str);

    printf("Enter a character: ");

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == ch)
