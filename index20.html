Q1:Write a program to show that example of pre-increment ?

#include 

int main()
{
    int a = 10;

    printf("Before increment = %d\n", a);

    printf("After pre-increment = %d\n", ++a);

    return 0;
}
Q2:Write a program to show that example of post-increment ? 

#include 

int main()
{
    int a = 10;

    printf("Before increment = %d\n", a);

    printf("After post-increment = %d\n", a++);

    printf("Final value = %d\n", a);

    return 0;
}

Q5:Write a program to enter a number from the user ,the length of the number is depend on the user ,programmer should print the last digit of number ? 

#include 

int main()
{
    int n, last;

    printf("Enter a number: ");
    scanf("%d", &n);

    last = n % 10;

    printf("Last digit = %d", last);

    return 0;
}
Q6:-Enter a decimal number from the user (float) and print the right hand side of the decimal point ? 

#include 

int main()
{
    float n;
    int decimal;

    printf("Enter a decimal number: ");
    scanf("%f", &n);

    decimal = (int)(n * 10) % 10;

    printf("Right hand side digit = %d", decimal);

    return 0;
}
