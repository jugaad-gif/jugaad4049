Question-3:-keep it simple Print all ASCII character 0 to 127 ? c language

#include 

int main() {
    int i;

    for (i = 0; i <= 127; i++) {
        printf("%d = %c\n", i, i);
    }

    return 0;
}




Question-4:-Enter two character from user and print ASCII value ?

#include 

int main() {
    char a, b;

    printf("Enter two characters: ");
    scanf(" %c %c", &a, &b);

    printf("ASCII value of %c = %d\n", a, a);
    printf("ASCII value of %c = %d\n", b, b);

    return 0;
}


Question-5:-Enter password display like ****** ?

#include 

int main() {
    char password[50];
    int i;

    printf("Enter password: ");
    scanf("%s", password);

    printf("Password: ");

    for (i = 0; password[i] != '\0'; i++) {

    }
