Question-2:-Write a program to enter 10 number from the user add the number and display the answer ? 
 #include 

 int main() {
    int i, num, sum = 0;

    for (i = 1; i <= 10; i++) {
        printf("Enter number %d: ", i);
        scanf("%d", &num);
        sum = sum + num;
    }

    printf("Sum = %d", sum);

    return 0;
}

Swap numbers using Call by address ? 
#include 

void swap(int *a, int *b) {
    int temp;
    temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    int x, y;

    printf("Enter two numbers: ");
    scanf("%d %d", &x, &y);

    swap(&x, &y);

    printf("After swapping: %d %d", x, y);

    return 0;
}
